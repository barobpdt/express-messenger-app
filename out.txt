/**
 * 태그 연결 상태 백업 (다운로드)
 * GET /api/esl/download-backup
 */
router.get('/download-backup', async (req, res) => {
    try {
        const workbook = new ExcelJS.Workbook();
        const worksheet = workbook.addWorksheet('Sheet1');

        worksheet.addRow(['ARTICLE_ID', 'LABEL_CODE', 'TEMPLATE_NAME']);
        
        const apiUrl = `${process.env.ESLURL}${process.env.DASHBOARDAPI}/common/labels?store=${process.env.STORE_CODE}&size=999&sort=code,ASC&page=0`;
        const response = await fetch(apiUrl, { method: 'GET', headers: { 'Content-Type': 'application/json' }});
        const data = await response.json();

        if (data.responseCode === "200" || data.responseCode === 200) {
            if (data.labelList) {
                data.labelList.forEach(label => {
                    if (label.articleList) {
                        label.articleList.forEach(article => {
                            worksheet.addRow([article.articleId, label.labelCode, ""]);
                        });
                    }
                });
            }
        } else {
            return res.status(400).json({ success: false, message: `태그 조회 실패 : ${JSON.stringify(data)}` });
        }

        worksheet.columns.forEach(column => { column.width = 20; });

        res.setHeader('Content-Type', 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet');
        res.setHeader('Content-Disposition', 'attachment; filename="before_unassign_backup.xlsx"');

        await workbook.xlsx.write(res);
        res.end();
    } catch (error) {
        res.status(500).json({ success: false, message: `에러 발생 : ${error.message}` });
    }
});

/**
 * 일괄 Unassign (unlink)
 * POST /api/esl/unassign
 */
router.post('/unassign', async (req, res) => {
    try {
        // 1. 전체 라벨 조회
        const getLabelsUrl = `${process.env.ESLURL}${process.env.DASHBOARDAPI}/common/labels?store=${process.env.STORE_CODE}`;
        const getResponse = await fetch(getLabelsUrl, { method: 'GET', headers: { 'Content-Type': 'application/json' }});
        const getData = await getResponse.json();

        const resultList = [];
        if (getData.responseCode === "200" || getData.responseCode === 200) {
            if (getData.labelList) {
                getData.labelList.forEach(label => {
                    resultList.push(label.labelCode);
                });
            }
        } else {
            return res.status(400).json({ success: false, message: `태그 조회 실패 : ${JSON.stringify(getData)}` });
        }

        // 2. Unassign 요청
        const unassignPayload = JSON.stringify({ unAssignList: resultList });
        const unassignUrl = `${process.env.ESLURL}${process.env.DASHBOARDAPI}/common/labels/unlink?store=${process.env.STORE_CODE}`;
        
        const unassignResponse = await fetch(unassignUrl, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: unassignPayload });
        const unassignData = await unassignResponse.json();

        if (unassignData.responseCode === "200" || unassignData.responseCode === 200) {
            res.json({ success: true, message: 'UnAssign 성공하였습니다.' });
        } else {
            res.status(400).json({ success: false, message: `Unassign 실패 : ${JSON.stringify(unassignData)}` });
        }
    } catch (error) {
        res.status(500).json({ success: false, message: `에러 발생 : ${error.message}` });
    }
});

/**
 * Generic API Proxy to bypass CORS and hit Solum Dashboard Services
 * POST /api/esl/proxy
 */
router.post('/proxy', async (req, res) => {
    try {
        const { method, path: apiPath, query, body } = req.body;
        
        // Build target URL
        const params = new URLSearchParams(query || {});
        const targetUrl = `${process.env.ESLURL}${process.env.DASHBOARDAPI}${apiPath}?${params.toString()}`;
        
        console.log(`[Proxy Request] ${method || 'GET'} -> ${targetUrl}`);
        
        const fetchOptions = {
            method: method || 'GET',
            headers: {
                'Content-Type': 'application/json;charset=UTF-8'
            }
        };

        if (method && method !== 'GET' && method !== 'HEAD' && body) {
            fetchOptions.body = JSON.stringify(body);
        }

        const response = await fetch(targetUrl, fetchOptions);
        
        // Handle response types
        const contentType = response.headers.get('content-type');
        let data;
        
        if (contentType && contentType.includes('application/json')) {
            data = await response.json();
        } else {
            data = { message: await response.text() };
        }
        
        res.status(response.status).json(data);
    } catch (error) {
        console.error('[Proxy Request Error]:', error);
        res.status(500).json({ success: false, message: error.message });
    }
});

        if (api.method === 'GET' || (api.queryParams && api.queryParams.length > 0)) {
            switchTab('params');
        } else {
            switchTab('body');
        }

        // Reset Response Panel
        responseMetaInfo.style.display = 'none';
        responseBodyOutput.textContent = '// Select API and click Send to execute...';
    }

    // 4. Render Query Parameters Table
    function renderQueryParameters(params = []) {
        paramsTableBody.innerHTML = '';
        params.forEach(param => {
            addParamRow(param.key, param.value, param.description);
        });
        updateParamsCount();
    }

    function addParamRow(key = '', value = '', desc = '') {
        const tr = document.createElement('tr');
        tr.className = 'param-row';
        tr.innerHTML = `
            <td><input type="text" class="param-key" value="${key}" placeholder="key"></td>
            <td><input type="text" class="param-value" value="${value}" placeholder="value"></td>
            <td><input type="text" class="param-desc" value="${desc}" placeholder="description"></td>
            <td>
                <button class="btn-delete-row"><i class="fa-solid fa-xmark"></i></button>
            </td>
        `;

        // Bind delete row action
        tr.querySelector('.btn-delete-row').addEventListener('click', () => {
            tr.remove();
            updateParamsCount();
        });

        // Bind value keyup to update badges
        tr.querySelectorAll('input').forEach(input => {
            input.addEventListener('keyup', updateParamsCount);
        });

        paramsTableBody.appendChild(tr);
        updateParamsCount();
    }

    function updateParamsCount() {
        const rows = paramsTableBody.querySelectorAll('.param-row');
        let count = 0;
        rows.forEach(row => {
            const key = row.querySelector('.param-key').value.trim();
            if (key) count++;
        });
        paramsCountBadge.textContent = count;
    }

    // 5. Handle Tabs Navigation
    function switchTab(tabId) {
        tabButtons.forEach(btn => {
            if (btn.getAttribute('data-tab') === tabId) {
                btn.classList.add('active');
            } else {
                btn.classList.remove('active');
            }
        });

        tabContents.forEach(content => {
            if (content.getAttribute('id') === `tab-${tabId}`) {
                content.classList.add('active');
            } else {
                content.classList.remove('active');
            }
        });
    }

    tabButtons.forEach(btn => {
        btn.addEventListener('click', () => {
            switchTab(btn.getAttribute('data-tab'));
        });
    });

    // 6. JSON formatting & Validation
    function validateJsonInput() {
        const val = requestBodyEditor.value.trim();
        if (!val) {
            jsonValMsg.textContent = '';
            return true;
        }

        try {
            JSON.parse(val);
            jsonValMsg.textContent = 'JSON syntax is valid.';
            jsonValMsg.className = 'json-validation-msg json-val-success';
            return true;
        } catch (e) {
            jsonValMsg.textContent = 'Invalid JSON: ' + e.message;
            jsonValMsg.className = 'json-validation-msg json-val-error';
            return false;
        }
    }

    requestBodyEditor.addEventListener('input', validateJsonInput);

    btnFormatJson.addEventListener('click', () => {
        const val = requestBodyEditor.value.trim();
        if (!val) return;
        try {
            const parsed = JSON.parse(val);
            requestBodyEditor.value = JSON.stringify(parsed, null, 4);
            validateJsonInput();
        } catch (e) {
            alert('Cannot format invalid JSON.');
        }
    });

    // 7. Console Logging Utility
    function appendLog(method, path, status, duration) {
        const timestamp = new Date().toTimeString().split(' ')[0];
        const logLine = document.createElement('div');
        logLine.className = 'console-line';

        let methodClass = 'c-method-get';
        if (method === 'POST') methodClass = 'c-method-post';
        if (method === 'PUT') methodClass = 'c-method-put';
        if (method === 'DELETE') methodClass = 'c-method-delete';

        const statusClass = status >= 200 && status < 300 ? 'c-status-success' : 'c-status-error';

        logLine.innerHTML = `
            <span class="timestamp">[${timestamp}]</span>
            <span class="c-method ${methodClass}">${method}</span>
            <span class="c-path">${path}</span>
            <span class="c-status ${statusClass}">${status}</span>
            <span class="c-time">${duration.toFixed(0)}ms</span>
        `;

        consoleLogsList.appendChild(logLine);
        consoleLogsList.scrollTop = consoleLogsList.scrollHeight; // Auto-scroll
    }

    btnClearConsole.addEventListener('click', () => {
        consoleLogsList.innerHTML = '<div class="console-line system-msg">Console cleared.</div>';
    });

    // 8. Execute HTTP API Proxy Call
    async function sendApiRequest() {
        const method = activeApi.method;
        const path = activeApi.path;

        // Parse Query Params Table
        const query = {};
        const rows = paramsTableBody.querySelectorAll('.param-row');
        rows.forEach(row => {
            const key = row.querySelector('.param-key').value.trim();
            const value = row.querySelector('.param-value').value.trim();
            if (key) {
                query[key] = value;
            }
        });

        // Parse Request Body JSON
        let body = null;
        if (!activeApi.isFileUpload && activeApi.bodyTemplate) {
            const bodyStr = requestBodyEditor.value.trim();
            if (bodyStr) {
                try {
                    body = JSON.parse(bodyStr);
                } catch (e) {
                    alert('Invalid JSON request body. Please fix syntax errors first.');
                    return;
                }
            }
        }

        // Show Loading State
        btnSendRequest.disabled = true;
        btnSendRequest.innerHTML = `<i class="fa-solid fa-spinner fa-spin"></i> Sending...`;
        responseBodyOutput.textContent = '// Executing API request...';
        responseMetaInfo.style.display = 'none';

        const startTime = performance.now();

        try {
            let response;
            const searchParams = new URLSearchParams(query);
            const queryStr = searchParams.toString();
            const url = path + (queryStr ? '?' + queryStr : '');

            if (path.startsWith('/api/')) {
                if (activeApi.isFileUpload) {
                    const file = requestFileInput.files[0];
                    if (!file) {
                        alert('Please select a file to upload first.');
                        btnSendRequest.disabled = false;
                        btnSendRequest.innerHTML = `<i class="fa-solid fa-paper-plane"></i> <span data-i18n="btn_send">Send Request</span>`;
                        return;
                    }
                    const formData = new FormData();
                    formData.append('file', file);
                    
                    response = await fetch(url, {
                        method: 'POST',
                        body: formData
                    });
                } else {
                    const fetchOptions = {
                        method: method,
                    };
                    if (method !== 'GET' && method !== 'HEAD') {
                        const bodyStr = requestBodyEditor.value.trim();
                        if (bodyStr) {
                            fetchOptions.headers = { 'Content-Type': 'application/json' };
                            fetchOptions.body = bodyStr;
                        }
                    }
                    response = await fetch(url, fetchOptions);
                }
            } else {
                // Target our local express proxy endpoint
                response = await fetch('/api/esl/proxy', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json'
                    },
                    body: JSON.stringify({
                        method,
                        path,
                        query,
                        body
                    })
                });
            }

            const duration = performance.now() - startTime;
            const status = response.status;

            // Load and display response (Handle binary vs JSON)
            let responseData;
            const contentType = response.headers.get('content-type');
            const isExcel = contentType && (
                contentType.includes('spreadsheetml') || 
                contentType.includes('octet-stream') ||
                contentType.includes('excel')
            );

            if (response.ok && (isExcel || activeApi.isFileDownload)) {
                const blob = await response.blob();
                const downloadUrl = window.URL.createObjectURL(blob);
                const a = document.createElement('a');
                a.href = downloadUrl;
                
                // Get filename
                let filename = 'downloaded_file.xlsx';
                const disposition = response.headers.get('content-disposition');
                if (disposition && disposition.indexOf('attachment') !== -1) {
                    const filenameRegex = /filename[^;=\n]*=((['"]).*?\2|[^;\n]*)/;
                    const matches = filenameRegex.exec(disposition);
                    if (matches != null && matches[1]) { 
                        filename = matches[1].replace(/['"]/g, '');
                    }
                } else {
                    if (path.includes('assign')) filename = 'assign.xlsx';
                    else if (path.includes('product')) filename = 'product.xlsx';
                    else if (path.includes('backup')) filename = 'backup.xlsx';
                }
                
                a.download = filename;
                document.body.appendChild(a);
                a.click();
                a.remove();
                window.URL.revokeObjectURL(downloadUrl);

                responseData = {
                    message: "File downloaded successfully",
                    filename: filename,
                    sizeBytes: blob.size,
                    contentType: contentType
                };
                responseBodyOutput.textContent = JSON.stringify(responseData, null, 4);
            } else {
                try {
                    responseData = await response.json();
                } catch (err) {
                    try {
                        const text = await response.text();
                        responseData = { message: text };
                    } catch (e2) {
                        responseData = { error: 'Failed to parse response', details: err.message };
                    }
                }
                responseBodyOutput.textContent = JSON.stringify(responseData, null, 4);
            }

            // Update status badge metadata
            resStatusBadge.textContent = `${status} ${response.statusText || (status === 200 ? 'OK' : 'Error')}`;
            resStatusBadge.className = `meta-badge ${status >= 200 && status < 300 ? 'status-success' : 'status-error'}`;
            resTimeBadge.innerHTML = `<i class="fa-regular fa-clock"></i> ${duration.toFixed(0)}ms`;
            responseMetaInfo.style.display = 'flex';

            // Append execution detail to console
            appendLog(method, path, status, duration);

        } catch (error) {
            const duration = performance.now() - startTime;
            responseBodyOutput.textContent = JSON.stringify({ error: 'Network Connection Failed', message: error.message }, null, 4);

            resStatusBadge.textContent = 'Connection Fail';
            resStatusBadge.className = 'meta-badge status-error';
            resTimeBadge.innerHTML = `<i class="fa-regular fa-clock"></i> ${duration.toFixed(0)}ms`;
            responseMetaInfo.style.display = 'flex';

            appendLog(method, path, 0, duration);
        } finally {
            // Restore Send Button
            btnSendRequest.disabled = false;
            btnSendRequest.innerHTML = `<i class="fa-solid fa-paper-plane"></i> <span data-i18n="btn_send">Send Request</span>`;

            // Reapply i18n
            if (window.setLanguage && typeof window.currentLang !== 'undefined') {
                window.setLanguage(window.currentLang);
            }
        }
    }

    // Bind event handlers
    btnSendRequest.addEventListener('click', sendApiRequest);
    btnAddParamRow.addEventListener('click', () => addParamRow('', '', ''));
    apiSearchInput.addEventListener('input', (e) => renderApiTree(e.target.value));

    // Handle File Input Change
    if (requestFileInput) {
        requestFileInput.addEventListener('change', (e) => {
            const file = e.target.files[0];
            if (file) {
                selectedFileName.textContent = file.name;
            } else {
                selectedFileName.textContent = 'No file selected';
            }
        });
    }

    // Toggle Console Card visibility
    const btnToggleConsole = document.getElementById('btn-toggle-console');
    const consoleCardPanel = document.getElementById('console-card-panel');
    const consoleToggleIcon = document.getElementById('console-toggle-icon');
    const consoleToggleText = document.getElementById('console-toggle-text');

    btnToggleConsole.addEventListener('click', () => {
        const isCollapsed = consoleCardPanel.classList.toggle('collapsed');
        if (isCollapsed) {
            consoleToggleIcon.className = 'fa-solid fa-chevron-up';
            consoleToggleText.setAttribute('data-i18n', 'btn_show');
        } else {
            consoleToggleIcon.className = 'fa-solid fa-chevron-down';
            consoleToggleText.setAttribute('data-i18n', 'btn_hide');
        }
        // Trigger translations refresh to update text
        if (window.setLanguage && typeof window.currentLang !== 'undefined') {
            window.setLanguage(window.currentLang);
        }
    });

    // Initialize Page load
    selectApi(apiCollection[0]);
    renderApiTree();
});
